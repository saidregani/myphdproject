        .. controls ($(#1,#2)+(\hw-0.2*\tp,\hh-0.6)$) and ($(#1,#2)+(\hw-0.6*\tp,-0.6)$) .. ($(#1,#2)+(\hw+0.4,-0.6)$)
        .. controls ($(#1,#2)+(\hw-0.6*\tp,-0.6)$) and ($(#1,#2)+(\hw-0.2*\tp,-\hh-0.6)$) .. ($(#1,#2)+(\hw-\tp,-\hh-0.6)$)
        -- ($(#1,#2)+(-\hw+\tp,-\hh-0.6)$)
        .. controls ($(#1,#2)+(-\hw+0.2*\tp,-\hh-0.6)$) and ($(#1,#2)+(-\hw+0.6*\tp,-0.6)$) .. ($(#1,#2)+(-\hw-0.4,-0.6)$)
        .. controls ($(#1,#2)+(-\hw+0.6*\tp,-0.6)$) and ($(#1,#2)+(-\hw+0.2*\tp,\hh-0.6)$) .. cycle;
  \filldraw[fill=cream,draw=gold,line width=0.9pt]
        ($(#1,#2)+(-\hw+\tp,\hh)$) -- ($(#1,#2)+(\hw-\tp,\hh)$)
        .. controls ($(#1,#2)+(\hw-0.2*\tp,\hh)$) and ($(#1,#2)+(\hw-0.6*\tp,0)$) .. ($(#1,#2)+(\hw,0)$)
        .. controls ($(#1,#2)+(\hw-0.6*\tp,0)$) and ($(#1,#2)+(\hw-0.2*\tp,-\hh)$) .. ($(#1,#2)+(\hw-\tp,-\hh)$)
        -- ($(#1,#2)+(-\hw+\tp,-\hh)$)
        .. controls ($(#1,#2)+(-\hw+0.2*\tp,-\hh)$) and ($(#1,#2)+(-\hw+0.6*\tp,0)$) .. ($(#1,#2)+(-\hw,0)$)
        .. controls ($(#1,#2)+(-\hw+0.6*\tp,0)$) and ($(#1,#2)+(-\hw+0.2*\tp,\hh)$) .. cycle;
  \fill[gold] ($(#1,#2)+(-\hw,0)$) circle (0.55) ($(#1,#2)+(\hw,0)$) circle (0.55);}
